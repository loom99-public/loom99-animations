```tsx
// hooks/useFontLoader.ts
import { useEffect, useState } from 'react';

declare global {
  interface Window {
    opentype: any;
  }
}

export function useFontLoader() {
  const [fontCache] = useState(new Map());

  const loadFont = async (fontFile: string) => {
    if (fontCache.has(fontFile)) {
      return fontCache.get(fontFile);
    }

    const font = await window.opentype.load(`fonts/${fontFile}.ttf`);
    fontCache.set(fontFile, font);
    return font;
  };

  return { loadFont, fontCache };
}
```
