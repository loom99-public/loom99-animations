**2.2 Implement compiler** (`gallery/src/editor/compiler/blocks/domain/PhaseClock.ts`)

New implementation following spec semantics:

```typescript
import type { BlockCompiler, RuntimeCtx } from '../../types';

type SignalPhase = (tMs: number, ctx: RuntimeCtx) => number;
type SignalUnit = (tMs: number, ctx: RuntimeCtx) => number;

export const PhaseClockBlock: BlockCompiler = {
  type: 'PhaseClock',

  inputs: [
    { name: 'tIn', type: { kind: 'Signal:time' } },
  ],

  outputs: [
    { name: 'phase', type: { kind: 'Signal:phase' } },
    { name: 'u', type: { kind: 'Signal:unit' } },
  ],

  compile({ params, inputs }) {
    // Validation
    if (!inputs.tIn) {
      throw new Error('PhaseClock: tIn input required');
    }
    if (inputs.tIn.kind !== 'Signal:time') {
      throw new Error(`PhaseClock: tIn must be Signal<time>, got ${inputs.tIn.kind}`);
    }

    const periodSec = Number(params.period ?? 3.0);
    const mode = String(params.mode ?? 'loop');

    if (periodSec <= 0) {
      throw new Error('PhaseClock: period must be > 0');
    }

    const periodMs = periodSec * 1000;
    const tInSignal = inputs.tIn.value; // upstream time signal

    // Compute phase from upstream time
    const phaseSignal: SignalPhase = (tMs, ctx) => {
      const t = tInSignal(tMs, ctx); // get unbounded time from upstream
      const raw = t / periodMs; // raw phase (unbounded)

      switch (mode) {
        case 'once':
          return Math.max(0, Math.min(1, raw));

        case 'pingpong': {
          const p = raw;
          const q = (p % 2 + 2) % 2; // handle negatives correctly
          return q < 1 ? q : 2 - q;
        }

        case 'loop':
        default:
          return (raw % 1 + 1) % 1; // handle negatives correctly
      }
    };

    // u output: clamped [0,1] for all modes
    const uSignal: SignalUnit = (tMs, ctx) => {
      const phase = phaseSignal(tMs, ctx);

      switch (mode) {
        case 'once':
          return phase; // already clamped

        case 'pingpong':
          // For pingpong, u tracks the forward portion
          const t = tInSignal(tMs, ctx);
          const raw = t / periodMs;
          const cycle = Math.floor(raw);
          const isForward = cycle % 2 === 0;
          return isForward ? phase : (1 - phase);

        case 'loop':
        default:
          return phase; // loop phase is already [0,1)
      }
    };

    return {
      phase: { kind: 'Signal:phase', value: phaseSignal },
      u: { kind: 'Signal:unit', value: uSignal },
    };
  },
};
```

## Risks & Mitigations

### Risk 1: TypeDesc System Incompatibility
**Likelihood**: Medium
**Impact**: High - blocks type system if types don't work

**Mitigation**:
- Start with string notation `'Signal<time>'` if TypeDesc not ready
- File TODO to migrate when type system stable
- Validate against `portUtils.ts` type checking

### Risk 2: Breaking Existing Macros
**Likelihood**: High - macros use PhaseClock
**Impact**: Medium - macros fail to expand

**Mitigation**:
- Update macros to use `PhaseClockLegacy` temporarily
- Or update macros to wire new PhaseClock to TimeRoot
- Add macro expansion tests

### Risk 3: u Output Semantics Unclear
**Likelihood**: Medium - spec doesn't fully define u for pingpong
**Impact**: Low - can refine later

**Mitigation**:
- Implement best guess (forward portion of pingpong)
- Document assumption in code comments
- Add TODO to verify with spec author if needed
