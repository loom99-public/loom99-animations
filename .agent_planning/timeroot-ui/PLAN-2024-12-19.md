# TimeRoot UI - Sprint Plan

**Sprint Goal**: Replace linear timeline with mode-aware Time Console (Finite mode only)

**Date**: 2024-12-19
**Estimated Complexity**: HIGH (3 deliverables, UI refactor + new component)
**Dependencies**: `timemodel-types` plan complete

---

## Sprint Goal

Replace the player's linear timeline scrubber with a TimeConsole component that displays different controls based on `TimeModel.kind`. Implement Finite mode UI fully, stub Cyclic/Infinite for future sprints.

**Why this matters**: Current UI lies to users by showing bounded timelines for infinite patches and looping toggles for structurally cyclic animations. The Time Console makes time topology explicit and honest.

## Technical Design

### TimeConsole Component API

```typescript
// TimeConsole.tsx
interface TimeConsoleProps {
  timeModel: TimeModel;
  currentTime: number; // tMs
  playState: PlayState; // 'playing' | 'paused'
  speed: number;
  seed: number;
  cuePoints: readonly CuePoint[];

  // Callbacks
  onScrub: (tMs: number) => void;
  onPlay: () => void;
  onPause: () => void;
  onSpeedChange: (speed: number) => void;
  onSeedChange: (seed: number) => void;
}

export const TimeConsole: React.FC<TimeConsoleProps> = (props) => {
  const { timeModel, playState, speed, seed, ...rest } = props;

  return (
    <div className="time-console">
      {/* Mode Badge */}
      <div className="time-console-header">
        <span className={`mode-badge mode-${timeModel.kind}`}>
          {timeModel.kind.toUpperCase()}
        </span>
      </div>

      {/* Mode-Specific Controls */}
      <div className="time-console-body">
        {timeModel.kind === 'finite' && (
          <FiniteControls
            durationMs={timeModel.durationMs}
            currentTime={props.currentTime}
            onScrub={props.onScrub}
            cuePoints={props.cuePoints}
          />
        )}
        {timeModel.kind === 'cyclic' && (
          <div className="mode-placeholder">CYCLE mode (coming soon)</div>
        )}
        {timeModel.kind === 'infinite' && (
          <div className="mode-placeholder">INFINITE mode (coming soon)</div>
        )}
      </div>

      {/* Always-Present Controls */}
      <div className="time-console-controls">
        <button onClick={playState === 'playing' ? props.onPause : props.onPlay}>
          {playState === 'playing' ? 'FREEZE' : 'RUN'}
        </button>

        <label>
          Speed
          <input type="number" value={speed} onChange={(e) => props.onSpeedChange(+e.target.value)} />
        </label>

        <label>
          Seed
          <input type="number" value={seed} onChange={(e) => props.onSeedChange(+e.target.value)} />
        </label>
      </div>
    </div>
  );
};
```

### FiniteControls Component API

```typescript
// FiniteControls.tsx
interface FiniteControlsProps {
  durationMs: number;
  currentTime: number;
  onScrub: (tMs: number) => void;
  cuePoints: readonly CuePoint[];
}

export const FiniteControls: React.FC<FiniteControlsProps> = ({
  durationMs,
  currentTime,
  onScrub,
  cuePoints,
}) => {
  const progress = durationMs > 0 ? (currentTime / durationMs) * 100 : 0;

  return (
    <div className="finite-controls">
      {/* Time Readouts */}
      <div className="finite-readouts">
        <span>Time: {formatTime(currentTime)} / {formatTime(durationMs)}</span>
        <span>Progress: {progress.toFixed(0)}%</span>
      </div>

      {/* Bounded Scrubber */}
      <div className="finite-scrubber">
        <span className="finite-time-start">{formatTime(0)}</span>

        <div className="finite-scrubber-container">
          <input
            type="range"
            min={0}
            max={durationMs}
            step={16}
            value={currentTime}
            onChange={(e) => onScrub(+e.target.value)}
          />

          {/* Cue Point Markers */}
          {cuePoints.map((cue, i) => {
            const percent = (cue.tMs / durationMs) * 100;
            return (
              <div
                key={i}
                className="cue-marker"
                style={{ left: `${percent}%` }}
                title={cue.label}
              />
            );
          })}
        </div>

        <span className="finite-time-end">{formatTime(durationMs)}</span>
      </div>
    </div>
  );
};

function formatTime(ms: number): string {
  return `${(ms / 1000).toFixed(2)}s`;
}
```
