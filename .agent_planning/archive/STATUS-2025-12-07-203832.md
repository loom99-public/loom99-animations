```javascript
async function loadFont(fontFile) {
    if (fontCache.has(fontFile)) {
        return fontCache.get(fontFile);
    }

    try {
        const font = await opentype.load(`fonts/${fontFile}.ttf`);
        fontCache.set(fontFile, font);
        return font;
    } catch (error) {
        console.error(`Failed to load font ${fontFile}:`, error);
        return null;
    }
}
```

Lines 1873-1924: Text-to-SVG path conversion
```javascript
function textToSvgPaths(font, text, fontSize = 80) {
    const words = text.split(/\s+/).filter(w => w.length > 0);
    const pathData = [];
    const colors = ['#ff2d2d', '#ff6b00', '#ffdd00', '#00d4ff', '#7b2ff7'];

    let currentY = 80;

    words.forEach((word, wordIndex) => {
        const path = font.getPath(word, 50, currentY, fontSize);
        const bounds = path.getBoundingBox();

        pathData.push({
            d: path.toPathData(2),
            text: word,
            color: colors[wordIndex % colors.length],
            y: currentY,
            bounds: bounds
        });

        currentY += fontSize * 1.2; // Line spacing
    });

    return pathData;
}
```

```javascript
}
```

```javascript
function createPathElements(pathData) {
    // Clear existing paths
    textGroup.innerHTML = '';
    pathElements = [];

    pathData.forEach((item, index) => {
        const path = document.createElementNS('http://www.w3.org/2000/svg', 'path');
        path.setAttribute('d', item.d);
        path.setAttribute('class', 'text-path');
        path.style.color = item.color;
        path.style.fill = item.color;
        path.style.filter = `drop-shadow(0 0 10px ${item.color})`;
        path.dataset.index = index;
        path.dataset.delay = index * 200; // Stagger delay
        textGroup.appendChild(path);
        pathElements.push({
            element: path,
            delay: index * 200,
            color: item.color
        });
    });
}
```
