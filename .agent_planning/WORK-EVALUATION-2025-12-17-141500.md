   ```typescript
   export function getBlockDefinition(type: string): BlockDefinition | undefined {
     // First try base blocks
     const baseDef = getBlockDefinitions().find((b) => b.type === type);
     if (baseDef) return baseDef;
     
     // Then try composites if type starts with 'composite:'
     if (type.startsWith('composite:')) {
       return getBlockDefinitions(true).find((b) => b.type === type);
     }
     return undefined;
   }
   ```
